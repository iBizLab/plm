```sql
SELECT
t1.CONFIDENCE,
t1.CONTEXT,
t1.CREATE_MAN,
t1.CREATE_TIME,
t1.DESCRIPTION,
t1.DOCUMENT_ID,
t21.NAME AS DOCUMENT_NAME,
t1.ID,
t1.KB_ID,
t11.NAME AS KB_NAME,
t1.KEYWORDS,
t1.NAME,
t1.NORMALIZED_NAME,
t1.REFERENCE_TYPE,
t1.TYPE,
t1.UPDATE_MAN,
t1.UPDATE_TIME
FROM AI_KB_GRAPH_ENTITY t1 
LEFT JOIN AI_KNOWLEDGE_BASE t11 ON t1.KB_ID = t11.ID 
LEFT JOIN AI_KB_DOCUMENT t21 ON t1.DOCUMENT_ID = t21.ID 

WHERE ( t1.KB_ID = #{ctx.datacontext.ai_knowledge_base} )
```
