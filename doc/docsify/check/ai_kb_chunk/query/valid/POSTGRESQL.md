SELECT
t1.ACTIVE,
t1.CONTENT,
t1.CONTENT_PREVIEW,
t1.CREATE_MAN,
t1.CREATE_TIME,
t1.DOCUMENT_ID,
t11.NAME AS DOCUMENT_NAME,
